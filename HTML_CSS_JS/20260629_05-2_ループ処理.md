## ★連想配列を使った繰り返し処理
### 1.for文［p.069］
［配列］
```js
let array = [10,20.5,'ABC'];

for (let i = 0; i < array.length; i++){
    console.log(i+'番目の要素:'+array[i]);
}
```
［連想配列］［補足］
```js
let array = {name:'太郎',age:21};
let keys = Object.keys(array) //キーの一覧の取得
for (let i = 0; i < array.length; i++){
    console.log('項目:'+keys[i]+'、'+'内容：'+array[keys[i]])
}
```

### 2.拡張for文［p.073］
```js
let array = [10,20.5,'ABC'];

//for (let 値を格納する変数 of 配列){
for (let i of array){}
    console.log(i;)
}
```
［連想配列］［補足］
```js
let array = {name:'太郎',age:21};
//for (let キーを格納する変数 in 連想配列){
for (let i in array){
    console.log('項目:'+ i +'、'+'内容：'+array[i]);
}
```
