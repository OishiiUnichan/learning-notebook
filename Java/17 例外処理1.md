```java
package code17_03;

import java.io.FileWriter;

public class Main {

	public static void main(String[] args) {
		try {
			FileWriter fw = new FileWriter("data.txt");
			fw.write("hello!");
			fw.close();
			System.out.println("ファイルに書き込みました");
		} 
	cacth(IOException e){System.out.println("失敗");}
	cacth(NullPointerException e){System.out.println("nullやぞ");}
/*
*例外発生時、記述した順番に検索し、最初にキャッチできるcacthブロックに処理以降するため、
*記述順番に注意が必要
*/
//同じ処理で対応する
	cacth(IOException | NullpointerException e){System.out.println("例外発生");}
}
//ざっくり例外キャッチ
catch(Exception e){("なんかエラー出た");}
```
